    function getCSRFToken() {
        const token = document.querySelector('[name=csrfmiddlewaretoken]');
        if (token && token.value) return token.value;
        
        const metaToken = document.querySelector('meta[name="csrf-token"]');
        if (metaToken) return metaToken.getAttribute('content');
        
        const cookie = document.cookie.split('; ').find(row => row.startsWith('csrftoken='));
        if (cookie) return cookie.split('=')[1];
        
        const allInputs = document.querySelectorAll('input[type=hidden]');
        for (let input of allInputs) {
            if (input.name && input.name.toLowerCase().includes('csrf')) {
                return input.value;
            }
        }
        
        return '';
    }

    function showMessage(message, type = 'success') {
        const oldMsg = document.querySelector('.account-toast');
        if (oldMsg) oldMsg.remove();
        
        const toast = document.createElement('div');
        toast.className = 'account-toast';
        
        let bgColor, icon;
        if (type === 'success') {

            icon = 'fa-check-circle';
        } else if (type === 'error') {
            bgColor = '#ef4444';
            icon = 'fa-exclamation-circle';
        } else {
            bgColor = '#f59e0b';
            icon = 'fa-exclamation-triangle';
        }
        
        toast.style.cssText = `
            position: fixed;
            bottom: 30px;
            right: 30px;
            padding: 14px 24px;
            background: ${bgColor};
            color: white;
            border-radius: 12px;
            z-index: 999999;
            font-size: 14px;
            font-weight: 500;
            font-family: system-ui, sans-serif;

        toast.innerHTML = `<i class="fas ${icon}" style="font-size: 18px;"></i> <span>${message}</span>`;
        document.body.appendChild(toast);
        
        toast.onclick = () => toast.remove();
        
        setTimeout(() => {

        }, 3000);

    function cleanPrice(priceText) {
        if (!priceText) return 0;
        if (typeof priceText === 'number') return priceText;
        
        let cleaned = priceText.toString();
        cleaned = cleaned.replace(/[$₼€£₺₽¥]/g, '');
        
        if (cleaned.includes(',') && cleaned.includes('.')) {
            cleaned = cleaned.replace(/,/g, '');
        } else if (cleaned.includes(',') && !cleaned.includes('.')) {
            const parts = cleaned.split(',');
            if (parts.length === 2 && parts[1].length === 2) {
                cleaned = parts[0] + '.' + parts[1];
            } else {
                cleaned = cleaned.replace(/,/g, '');
            }
        }
        
        cleaned = cleaned.replace(/[^0-9.-]/g, '');
        
        const parsed = parseFloat(cleaned);
        return isNaN(parsed) ? 0 : parsed;
    }

    // 7. SERVER SORĞUSU - DÜZGÜNLAŞDİRİLDİ
    // ============================================================
    async function sendRequest(action, data) {
        const formData = new FormData();
        const csrf = getCSRFToken();
        if (csrf) formData.append('csrfmiddlewaretoken', csrf);
        
        // BU ÖNƏMLİDİR - action parametrini birbaşa əlavə et
        formData.append('action', action);
        
        Object.keys(data).forEach(key => {
            if (data[key] !== undefined && data[key] !== null && data[key] !== '') {
                formData.append(key, data[key]);
            }
        });
        
        try {
            const response = await fetch(window.location.href, {
                method: 'POST',
                body: formData,
                headers: { 'X-Requested-With': 'XMLHttpRequest' }
            });
            
            const result = await response.json();
            console.log(`📦 ${action} response:`, result);
            return result;
        } catch (error) {
            console.error('Request error:', error);
            return { status: 'error', message: 'Network error!' };
        }
    }

    window.submitComment = async function(event) {
        if (event) event.preventDefault();
        
        const form = document.getElementById('commentForm');
        if (!form) {
            console.error('Comment form not found!');
            return;
        }
        
        const action = document.getElementById('actionInput')?.value || 'add_review';
        const textarea = document.getElementById('commentText');
        const content = textarea ? textarea.value.trim() : '';
        const firstName = document.getElementById('firstName')?.value.trim() || '';
        const lastName = document.getElementById('lastName')?.value.trim() || '';
        const fullName = `${firstName} ${lastName}`.trim();
        
        // Rating-i yoxla (əgər review action-dırsa)
        let rating = null;
        if (action === 'add_review') {
            const selectedRating = document.querySelector('input[name="rating"]:checked');
            rating = selectedRating ? parseInt(selectedRating.value) : null;
        }
        
        // Validasiya
        if (!content) {
            showMessage('Zəhmət olmasa şərh yazın!', 'error');
            return;
        }
        
        if (action === 'add_review' && !fullName) {
            showMessage('Zəhmət olmasa ad və soyadınızı daxil edin!', 'error');
            return;
        }
        
        const submitBtn = document.getElementById('submitBtn');
        const originalHtml = submitBtn?.innerHTML || 'Göndər';
        
        if (submitBtn) {
            submitBtn.disabled = true;
            submitBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Göndərilir...';
        }
        
        // Məlumatları hazırla
        const postData = {
            text: content,
            rating: rating
        };
        
        // Əgər review əlavə edirsə, ad məlumatlarını da göndər
        if (action === 'add_review') {
            postData.name = firstName;
            postData.surname = lastName;
        }
        
        // Cavab verirsə, parent_id məlumatını əlavə et
        const parentId = document.getElementById('parentId')?.value;
        if (parentId) {
            postData.parent_id = parentId;
        }
        
        const result = await sendRequest(action, postData);
        
        if (result.status === 'success') {
            // Yeni şərhi səhifəyə əlavə et
            if (result.review) {
                addCommentToPage(result.review);
            } else if (result.reply) {
                addReplyToPage(result.reply);
            }
            
            // Formu təmizlə
            if (textarea) textarea.value = '';
            if (rating !== null) {
                const radioButtons = document.querySelectorAll('input[name="rating"]');
                radioButtons.forEach(radio => radio.checked = false);
                const ratingText = document.getElementById('ratingText');
                if (ratingText) ratingText.textContent = '0/5';
            }
            
            // Ad sahələrini təmizləmə (istəyə bağlı)
            // document.getElementById('firstName').value = '';
            // document.getElementById('lastName').value = '';
            
            // Cavab məlumatlarını sıfırla
            resetReplyForm();
            
            showMessage(result.message || '✅ Şərh əlavə edildi!', 'success');
        } else {
            showMessage(result.message || '❌ Xəta baş verdi!', 'error');
        }
        
        if (submitBtn) {
            submitBtn.disabled = false;
            submitBtn.innerHTML = originalHtml;
        }
    };
    
    // Yeni cavabı səhifəyə əlavə et
    function addReplyToPage(reply) {
        if (!reply || !reply.parent_id) return;
        
        // Əsas şərhi tap
        const parentComment = document.querySelector(`.comment-item[data-comment-id="${reply.parent_id}"]`);
        if (!parentComment) return;
        
        // Replies container-ını tap və ya yarat
        let repliesContainer = parentComment.querySelector(`#replies-container-${reply.parent_id}`);
        if (!repliesContainer) {
            const repliesDiv = document.createElement('div');
            repliesDiv.className = 'comment-replies';
            repliesDiv.id = `replies-container-${reply.parent_id}`;
            repliesDiv.style.display = 'block';
            
            // Toggle button əlavə et
            const toggleBtn = document.createElement('button');
            toggleBtn.className = 'toggle-replies-btn';
            toggleBtn.setAttribute('onclick', `window.toggleReplies(${reply.parent_id})`);
            toggleBtn.innerHTML = `<i class="fas fa-comment-dots"></i> Cavabları gizlət (1)`;
            
            const commentContent = parentComment.querySelector('.comment-content');
            if (commentContent) {
                commentContent.appendChild(repliesDiv);
                commentContent.appendChild(toggleBtn);
            }
            repliesContainer = repliesDiv;
        }
        
        // Cavabı əlavə et
        const replyHtml = createReplyHTML(reply);
        repliesContainer.insertAdjacentHTML('beforeend', replyHtml);
        
        // Toggle button mətnini yenilə
        const toggleBtn = parentComment.querySelector('.toggle-replies-btn');
        if (toggleBtn) {
            const replyCount = repliesContainer.children.length;
            toggleBtn.innerHTML = `<i class="fas fa-comment-dots"></i> Cavabları gizlət (${replyCount})`;
        }
        
        // Cavabları göstər
        repliesContainer.style.display = 'block';
        
        // Yeni cavaba animasiya əlavə et
        const newReply = repliesContainer.querySelector(`.reply-item[data-reply-id="${reply.id}"]`);
        if (newReply) {
            newReply.style.animation = 'slideInUp 0.3s ease';
            setTimeout(() => {
                newReply.style.animation = '';
            }, 300);
        }
        
        // Şərh sayını yenilə (cavablar şərh sayına daxil deyilsə, bura lazım deyil)
    }
    
    // Şərhə cavab yazmağa hazırlaş
    window.prepareReply = function(commentId, userName) {
        const actionInput = document.getElementById('actionInput');
        const parentIdInput = document.getElementById('parentId');
        const replyToNameSpan = document.getElementById('replyToDisplayName');
        const replyInfo = document.getElementById('replyInfo');
        const ratingSection = document.getElementById('ratingSection');
        const commentText = document.getElementById('commentText');
        const submitBtn = document.getElementById('submitBtn');
        
        if (actionInput) actionInput.value = 'add_reply';
        if (parentIdInput) parentIdInput.value = commentId;
        if (replyToNameSpan) replyToNameSpan.textContent = userName;
        if (replyInfo) replyInfo.style.display = 'flex';
        
        // Rating bölməsini gizlət (cavabda rating olmur)
        if (ratingSection) ratingSection.style.display = 'none';
        
        // Submit button mətnini dəyiş
        if (submitBtn) {
            submitBtn.querySelector('span').textContent = 'Cavab göndər';
        }
        
        // Textarea-a fokusla
        if (commentText) {
            commentText.focus();
            commentText.placeholder = `${userName} -a cavab yazın...`;
        }
        
        // Səhifəni yumşaq şəkildə yuxarı çək
        commentText?.scrollIntoView({ behavior: 'smooth', block: 'center' });
    };

        if (ratingText) ratingText.textContent = '0/5';

    // Yeni şərhi səhifəyə əlavə et
    function addCommentToPage(review) {
        const commentsList = document.getElementById('commentsList');
        if (!commentsList) return;
        
        // Boş vəziyyət mesajını sil
        const emptyComments = document.getElementById('emptyComments');
        if (emptyComments) emptyComments.remove();
        
        const commentHtml = createCommentHTML(review);
        commentsList.insertAdjacentHTML('afterbegin', commentHtml);
        
        // Şərh sayını yenilə
        updateCommentCount();
        
        // Yeni şərhə animasiya əlavə et
        const newComment = commentsList.querySelector(`.comment-item[data-comment-id="${review.id}"]`);
        if (newComment) {
            newComment.style.animation = 'slideInUp 0.3s ease';
            setTimeout(() => {
                newComment.style.animation = '';
            }, 300);
        }
    }
    
    // Şərh HTML-i yarat
    function createCommentHTML(review) {
        const likeClass = review.user_liked ? 'liked' : '';
        const dislikeClass = review.user_disliked ? 'disliked' : '';
        const hasReplies = review.replies && review.replies.length > 0;
        const ratingHtml = review.rating ? `<div class="comment-rating"><i class="fas fa-star"></i> ${review.rating}/5</div>` : '';
        
        return `
            <div class="comment-item" data-comment-id="${review.id}">
                <div class="comment-avatar">
                    <img src="${review.avatar || '/static/images/default-avatar.png'}" alt="${escapeHtml(review.user_name)}">
                </div>
                <div class="comment-content">
                    <div class="comment-header">
                        <span class="comment-author">${escapeHtml(review.user_name)}</span>
                        ${ratingHtml}
                        <span class="comment-date">${review.created_at || 'İndi'}</span>
                    </div>
                    <div class="comment-text">${escapeHtml(review.text)}</div>
                    <div class="comment-actions">
                        <button class="comment-like-btn ${likeClass}" onclick="window.rateComment(${review.id}, 'like')">
                            <i class="fas fa-thumbs-up"></i>
                            <span class="like-count">${review.likes || 0}</span>
                        </button>
                        <button class="comment-dislike-btn ${dislikeClass}" onclick="window.rateComment(${review.id}, 'dislike')">
                            <i class="fas fa-thumbs-down"></i>
                            <span class="dislike-count">${review.dislikes || 0}</span>
                        </button>
                        <button class="comment-reply-btn" onclick="window.prepareReply(${review.id}, '${escapeHtml(review.user_name)}')">
                            <i class="fas fa-reply"></i> Cavab ver
                        </button>
                    </div>
                    ${hasReplies ? `
                        <div class="comment-replies" id="replies-container-${review.id}" style="display: block;">
                            ${review.replies.map(reply => createReplyHTML(reply)).join('')}
                        </div>
                        <button class="toggle-replies-btn" onclick="window.toggleReplies(${review.id})">
                            <i class="fas fa-comment-dots"></i> Cavabları gizlət (${review.replies.length})
                        </button>
                    ` : `
                        <div class="comment-replies" id="replies-container-${review.id}" style="display: none;"></div>
                    `}
                </div>
            </div>
        `;
    }
    
    // Cavab HTML-i yarat
    function createReplyHTML(reply) {
        const likeClass = reply.user_liked ? 'liked' : '';
        const dislikeClass = reply.user_disliked ? 'disliked' : '';
        
        return `
            <div class="reply-item" data-reply-id="${reply.id}">
                <div class="reply-avatar">
                    <img src="${reply.avatar || '/static/images/default-avatar.png'}" alt="${escapeHtml(reply.user_name)}">
                </div>
                <div class="reply-content">
                    <div class="reply-header">
                        <span class="reply-author">${escapeHtml(reply.user_name)}</span>
                        ${reply.parent_user_name ? `<span class="reply-to">→ ${escapeHtml(reply.parent_user_name)}</span>` : ''}
                        <span class="reply-date">${reply.created_at || 'İndi'}</span>
                    </div>
                    <div class="reply-text">${escapeHtml(reply.text)}</div>
                    <div class="reply-actions">
                        <button class="reply-like-btn ${likeClass}" onclick="window.rateComment(${reply.id}, 'like')">
                            <i class="fas fa-thumbs-up"></i>
                            <span>${reply.likes || 0}</span>
                        </button>
                        <button class="reply-dislike-btn ${dislikeClass}" onclick="window.rateComment(${reply.id}, 'dislike')">
                            <i class="fas fa-thumbs-down"></i>
                            <span>${reply.dislikes || 0}</span>
                        </button>
                        <button class="reply-reply-btn" onclick="window.prepareReply(${reply.id}, '${escapeHtml(reply.user_name)}')">
                            <i class="fas fa-reply"></i> Cavab ver
                        </button>
                    </div>
                </div>
            </div>
        `;
    }
    
    // Cavabları göstər/gizlət
    window.toggleReplies = function(commentId) {
        const repliesContainer = document.getElementById(`replies-container-${commentId}`);
        const toggleBtn = document.querySelector(`.comment-item[data-comment-id="${commentId}"] .toggle-replies-btn`);
        
        if (repliesContainer && toggleBtn) {
            if (repliesContainer.style.display === 'none' || repliesContainer.style.display === '') {
                repliesContainer.style.display = 'block';
                const replyCount = repliesContainer.children.length;
                toggleBtn.innerHTML = `<i class="fas fa-comment-dots"></i> Cavabları gizlət (${replyCount})`;
            } else {
                repliesContainer.style.display = 'none';
                const replyCount = repliesContainer.children.length;
                toggleBtn.innerHTML = `<i class="fas fa-comment-dots"></i> Cavabları göstər (${replyCount})`;
            }
        }
    };
    
    // Şərhə like/dislike et
    window.rateComment = async function(commentId, action) {
        const result = await sendRequest('rate_comment', {
            comment_id: commentId,
            action: action
        });
        
        if (result.status === 'success') {
            updateCommentRating(commentId, result.likes, result.dislikes, result.user_action);
            showMessage(result.message || '✅ Rəy qeydə alındı!', 'success');
        } else {
            showMessage(result.message || '❌ Xəta baş verdi!', 'error');
        }
    };
    
    // Şərh ratingini yenilə
    function updateCommentRating(commentId, likes, dislikes, userAction) {
        const commentElement = document.querySelector(`.comment-item[data-comment-id="${commentId}"], .reply-item[data-reply-id="${commentId}"]`);
        if (!commentElement) return;
        
        const likeBtn = commentElement.querySelector('.comment-like-btn, .reply-like-btn');
        const dislikeBtn = commentElement.querySelector('.comment-dislike-btn, .reply-dislike-btn');
        const likeCountSpan = likeBtn?.querySelector('span:not(.fa)');
        const dislikeCountSpan = dislikeBtn?.querySelector('span:not(.fa)');
        
        if (likeCountSpan) likeCountSpan.textContent = likes || 0;
        if (dislikeCountSpan) dislikeCountSpan.textContent = dislikes || 0;
        
        // Düymələrin stilini yenilə
        if (likeBtn && dislikeBtn) {
            if (userAction === 'like') {
                likeBtn.classList.add('liked');
                dislikeBtn.classList.remove('disliked');
            } else if (userAction === 'dislike') {
                likeBtn.classList.remove('liked');
                dislikeBtn.classList.add('disliked');
            } else {
                likeBtn.classList.remove('liked');
                dislikeBtn.classList.remove('disliked');
            }
        }
    }
    
    // Şərh sayını yenilə
    function updateCommentCount() {
        const commentsList = document.getElementById('commentsList');
        const commentCountSpan = document.getElementById('commentCount');
        const commentCountDisplay = document.getElementById('commentCountDisplay');
        
        if (commentsList && commentCountSpan) {
            const count = commentsList.querySelectorAll('.comment-item').length;
            commentCountSpan.textContent = count;
            if (commentCountDisplay) commentCountDisplay.textContent = count;
        }
    }
    
    // Daha çox şərh yüklə
    window.loadMoreComments = async function() {
        const loadMoreBtn = document.getElementById('loadMoreBtn');
        const currentCount = document.querySelectorAll('.comment-item').length;
        
        if (loadMoreBtn) {
            loadMoreBtn.disabled = true;
            loadMoreBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Yüklənir...';
        }
        
        const result = await sendRequest('load_more_comments', {
            offset: currentCount,
            limit: 5
        });
        
        if (result.status === 'success' && result.comments && result.comments.length > 0) {
            const commentsList = document.getElementById('commentsList');
            if (commentsList) {
                result.comments.forEach(comment => {
                    const commentHtml = createCommentHTML(comment);
                    commentsList.insertAdjacentHTML('beforeend', commentHtml);
                });
            }
            updateCommentCount();
        }
        
        if (result.has_more === false && loadMoreBtn) {
            loadMoreBtn.style.display = 'none';
        }
        
        if (loadMoreBtn) {
            loadMoreBtn.disabled = false;
            loadMoreBtn.innerHTML = '<span>Daha çox şərh yükləyin</span><i class="fas fa-rotate-right"></i>';
        }
    };

    window.updateProfileInfoAJAX = async function(submitBtn, event) {
        if (event && event.preventDefault) {
            event.preventDefault();
            event.stopPropagation();
        }
        
        const firstNameInput = document.querySelector('#first_name, [name="first_name"], #id_first_name');
        const lastNameInput = document.querySelector('#last_name, [name="last_name"], #id_last_name');
        const phoneInput = document.querySelector('#phone, [name="phone"], #id_phone');
        const birthdateInput = document.querySelector('#birthdate, [name="birthdate"], #id_birthdate, input[type="date"]');
        
        let gender = '';
        const genderRadio = document.querySelector('input[name="gender"]:checked');
        if (genderRadio) {
            gender = genderRadio.value;
        } else {
            gender = document.querySelector('#gender, [name="gender"], select[name="gender"]')?.value || '';
        }
        
        const receiveOffers = document.querySelector('#receive_offers, [name="receive_offers"]')?.checked || false;
        const subscribeNewsletter = document.querySelector('#subscribe_newsletter, [name="subscribe_newsletter"]')?.checked || false;
        
        const firstName = firstNameInput?.value || '';
        const lastName = lastNameInput?.value || '';
        const phone = phoneInput?.value || '';
        const birthdate = birthdateInput?.value || '';
        
        const btn = submitBtn || event?.target?.closest('button[type="submit"]');
        const originalHtml = btn?.innerHTML || 'Update';
        if (btn) {
            btn.disabled = true;
            btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Yenilənir...';
        }
        
        const result = await sendRequest('update_profile', {
            first_name: firstName,
            last_name: lastName,
            phone: phone,
            birthdate: birthdate,
            gender: gender,
            receive_offers: receiveOffers ? 'on' : '',
            subscribe_newsletter: subscribeNewsletter ? 'on' : ''
        });
        
        if (result.status === 'success') {
            updateAllNameDisplays(firstName, lastName);
            showMessage(result.message || '✅ Profil məlumatları yeniləndi!', 'success');
        } else {
            showMessage(result.message || '❌ Xəta baş verdi!', 'error');

            btn.disabled = false;
            btn.innerHTML = originalHtml;
        }

    window.removeFromCart = async function(productId, btn, skipConfirm = false) {
        if (!skipConfirm) {
            if (!confirm('Məhsulu səbətdən silmek istədiyinizə əminsiniz?')) return;
        }
        
        const row = btn?.closest('tr.cart-item');
        const originalHtml = btn?.innerHTML;
        if (btn) {
            btn.disabled = true;
            btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i>';
        }
        
        const result = await sendRequest('remove_from_cart', { product_id: productId });
        
        if (result.status === 'success') {
            if (row) {
                row.style.transition = 'all 0.3s ease';
                row.style.opacity = '0';
                setTimeout(() => {
                    row.remove();
                    updateBadges(result.cart_count, result.wishlist_count);
                    updateCartTotal();
                    showMessage('✅ Item removed from cart', 'success');
                    
                    const tbody = document.querySelector('#cart-table-body');

                        location.reload();
                    }

            } else {

            }
        } else {
            showMessage(result.message || 'Error!', 'error');
            if (btn) {
                btn.disabled = false;
                btn.innerHTML = originalHtml;
            }
        }
    };
    
    // SƏBƏTDƏ MİQDAR YENİLƏ
    window.updateQuantity = async function(btn, change, productId) {
        const row = btn?.closest('tr.cart-item');
        if (!row) {
            showMessage('Xəta: Sətir tapılmadı!', 'error');
            return;
        }
        
        const qtySpan = row.querySelector('.qty-value');
        if (!qtySpan) {
            showMessage('Xəta: Miqdar elementi tapılmadı!', 'error');
            return;
        }
        
        let unitPrice = null;
        if (row.dataset.unitPrice) {
            unitPrice = parseFloat(row.dataset.unitPrice);
        }
        
        if (!unitPrice || isNaN(unitPrice)) {
            const priceSpan = row.querySelector('.price');
            if (priceSpan) {
                const dataPrice = priceSpan.dataset.price;
                if (dataPrice) {
                    unitPrice = cleanPrice(dataPrice);
                } else {
                    unitPrice = cleanPrice(priceSpan.textContent);
                }
                
                const currentQty = parseInt(qtySpan.textContent) || 1;
                if (currentQty > 1 && unitPrice > 10000) {
                    unitPrice = unitPrice / currentQty;
                }
            }
        }
        
        if (!unitPrice || isNaN(unitPrice) || unitPrice <= 0) {
            const totalSpan = row.querySelector('.total-price');
            const currentQty = parseInt(qtySpan.textContent) || 1;
            if (totalSpan && currentQty > 0) {
                const totalValue = cleanPrice(totalSpan.textContent);
                unitPrice = totalValue / currentQty;
            }
        }
        
        if (!unitPrice || isNaN(unitPrice) || unitPrice <= 0) {
            showMessage('Qiymət məlumatı tapılmadı!', 'error');
            return;
        }
        
        if (!row.dataset.unitPrice) {
            row.dataset.unitPrice = unitPrice;
        }
        
        let currentQty = parseInt(qtySpan.textContent) || 1;
        let newQty = currentQty + change;
        
        if (newQty <= 0) {
            await window.removeFromCart(productId, btn, true);
            return;
        }
        
        const originalHtml = btn.innerHTML;
        btn.disabled = true;
        btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i>';
        
        const result = await sendRequest('update_cart_quantity', { 
            product_id: productId, 
            quantity: newQty 
        });
        
        if (result.status === 'success') {
            qtySpan.textContent = newQty;
            const qtyInput = row.querySelector('.quantity-input');
            if (qtyInput) qtyInput.value = newQty;
            const newTotal = unitPrice * newQty;
            const totalSpan = row.querySelector('.total-price');
            if (totalSpan) {
                totalSpan.textContent = formatPrice(newTotal);
            }
            updateCartTotal();
            updateBadges(result.cart_count, result.wishlist_count);
            showMessage(`✅ Miqdar: ${newQty} | Cəmi: ${formatPrice(newTotal)}`, 'success');
        } else {
            showMessage(result.message || 'Xəta baş verdi!', 'error');
            qtySpan.textContent = currentQty;
        }
        
        btn.disabled = false;
        btn.innerHTML = originalHtml;
    };
    
    // ÜMUMİ SƏBƏT CƏMİNİ YENİLƏ
    function updateCartTotal() {
        let total = 0;
        const allRows = document.querySelectorAll('#cart-table-body tr.cart-item');
        
        allRows.forEach(row => {
            const totalSpan = row.querySelector('.total-price');
            if (totalSpan) {
                const price = cleanPrice(totalSpan.textContent);
                if (!isNaN(price)) {
                    total += price;
                }
            }
        });
        
        const subtotalElement = document.querySelector('#cart-subtotal, .cart-subtotal, .subtotal-amount');
        if (subtotalElement) {
            subtotalElement.innerHTML = `Cəmi: ${formatPrice(total)}`;
        }
        
        const totalElements = document.querySelectorAll('.cart-grand-total, .total-amount, #cart-total, .cart-total');
        totalElements.forEach(el => {
            el.textContent = formatPrice(total);
        });
        
        return total;
    }

    function initWishlistSearch() {
        const searchInput = document.getElementById('wishlist-search');
        if (!searchInput) return;
        
        let debounceTimer;
        searchInput.addEventListener('input', function(e) {
            clearTimeout(debounceTimer);
            debounceTimer = setTimeout(() => {
                filterWishlistTable(this.value);
            }, 300);
        });
        
        function filterWishlistTable(searchTerm) {
            const tableBody = document.querySelector('#wishlist-table-body');
            if (!tableBody) return;
            
            const rows = tableBody.querySelectorAll('tr.wishlist-item');
            const searchLower = searchTerm.toLowerCase().trim();
            
            rows.forEach(row => {
                const productNameEl = row.querySelector('.product-name');
                let productName = productNameEl ? productNameEl.textContent.toLowerCase() : '';
                const matches = searchLower === '' || productName.includes(searchLower);
                row.style.display = matches ? '' : 'none';
            });
        }
    }
    
    function initCartSearch() {
        const searchInput = document.getElementById('cart-search');
        if (!searchInput) return;
        
        let debounceTimer;
        searchInput.addEventListener('input', function(e) {
            clearTimeout(debounceTimer);
            debounceTimer = setTimeout(() => {
                filterCartTable(this.value);
            }, 300);
        });
        
        function filterCartTable(searchTerm) {
            const tableBody = document.querySelector('#cart-table-body');
            if (!tableBody) return;
            
            const rows = tableBody.querySelectorAll('tr.cart-item');
            const searchLower = searchTerm.toLowerCase().trim();
            
            rows.forEach(row => {
                const productNameEl = row.querySelector('.product-name');

                const matches = searchLower === '' || productName.includes(searchLower);

            });
            
            updateCartTotal();
        }
    }

    function attachEventListeners() {
        console.log('🔄 Attaching event listeners...');
        
        // Comment formu
        const commentForm = document.getElementById('commentForm');
        if (commentForm && !commentForm.dataset._attached) {
            commentForm.dataset._attached = 'true';
            commentForm.addEventListener('submit', (e) => {
                e.preventDefault();
                window.submitComment(e);
            });
        }
        
        // Ləğv et düyməsi
        const cancelReplyBtn = document.getElementById('cancelReply');
        if (cancelReplyBtn && !cancelReplyBtn.dataset._attached) {
            cancelReplyBtn.dataset._attached = 'true';
            cancelReplyBtn.addEventListener('click', () => window.cancelReply());
        }
        
        // Daha çox yüklə düyməsi
        const loadMoreBtn = document.getElementById('loadMoreBtn');
        if (loadMoreBtn && !loadMoreBtn.dataset._attached) {
            loadMoreBtn.dataset._attached = 'true';
            loadMoreBtn.addEventListener('click', () => window.loadMoreComments());
        }
        
        // Rating ulduzları
        const starContainer = document.querySelector('.star-rating');
        if (starContainer && !starContainer.dataset._attached) {
            starContainer.dataset._attached = 'true';
            starContainer.addEventListener('mouseleave', function() {
                const checked = document.querySelector('input[name="rating"]:checked');
                const stars = document.querySelectorAll('.star-rating label');
                const ratingText = document.getElementById('ratingText');
                
                if (checked) {
                    const value = parseInt(checked.value);
                    for (let i = 0; i < stars.length; i++) {
                        if (stars.length - i <= value) {
                            stars[i].style.color = '#ffc107';
                        } else {
                            stars[i].style.color = '#ddd';
                        }
                    }
                    if (ratingText) ratingText.textContent = `${value}/5`;
                } else {
                    stars.forEach(star => star.style.color = '#ddd');
                    if (ratingText) ratingText.textContent = '0/5';
                }
            });
        }
        
        // Rating radio dəyişdikdə
        const ratingRadios = document.querySelectorAll('input[name="rating"]');
        ratingRadios.forEach(radio => {
            if (!radio.dataset._attached) {
                radio.dataset._attached = 'true';
                radio.addEventListener('change', function() {
                    const ratingText = document.getElementById('ratingText');
                    if (ratingText) ratingText.textContent = `${this.value}/5`;
                });
            }
        });
        
        // Profil yeniləmə düymələri
        const profileBtns = document.querySelectorAll('#update-profile-btn, .update-profile-btn, .btn-save-profile, .save-profile-btn, [data-action="update-profile"], #save-profile-btn');
        profileBtns.forEach(btn => {
            if (!btn.dataset._attached) {
                btn.dataset._attached = 'true';
                btn.addEventListener('click', (e) => {
                    e.preventDefault();
                    window.updateProfileInfoAJAX(btn, e);
                });
            }
        });
        
        initWishlistSearch();
        initCartSearch();
    }

    function escapeHtml(str) {
        if (!str) return '';
        return str.replace(/[&<>]/g, function(m) {
            if (m === '&') return '&amp;';
            if (m === '<') return '&lt;';
            if (m === '>') return '&gt;';
            return m;
        });
    }
